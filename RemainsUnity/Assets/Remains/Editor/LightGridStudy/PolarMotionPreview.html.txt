const controls={track:document.querySelector('#track'),filter:document.querySelector('#filter'),left:document.querySelector('#left'),right:document.querySelector('#right'),
play:document.querySelector('#play'),previous:document.querySelector('#previous'),next:document.querySelector('#next'),time:document.querySelector('#time'),speed:document.querySelector('#speed')};
const canvases=[document.querySelector('#a'),document.querySelector('#b')],status=document.querySelector('#status'),cache=new Map;
let report,frame=0,serial=0,playing=false,timer,zoom=1,cx=.5,cy=.5,current=[],currentUrls=[],clock=0,shown=[];
function bitmap(url){
 if(!cache.has(url)){
  const entry={last:++clock,value:null,promise:null};
  entry.promise=fetch(url).then(r=>{if(!r.ok)throw Error('Не загружен '+url+' ('+r.status+')');return r.blob()}).then(createImageBitmap).then(b=>{entry.value=b;return b});
  // Prefetch failures are retained, then shown explicitly if the frame is actually requested.
  entry.promise.catch(()=>{});cache.set(url,entry);
 }
 const entry=cache.get(url);entry.last=++clock;return entry.promise;
}
function trim(){
 const ready=[...cache].filter(([url,e])=>e.value&&!currentUrls.includes(url)).sort((a,b)=>a[1].last-b[1].last);
 while(cache.size>12&&ready.length){const [url,e]=ready.shift();e.value.close();cache.delete(url);}
}
function config(){
 const track=report.tracks.find(t=>t.id===controls.track.value);
 return {track,filter:track.flash?'fixed':controls.filter.value,left:track.flash?'aa':controls.left.value,right:track.flash?'05':controls.right.value};
}
function file(c,side,n){
 const variant=c[side]==='aa'?'aa':'p'+c[side]+'-'+c.filter;
 const index=c.track.flash?(n===12||n===36?1:0):n;
 return 'motion-'+c.track.id+'-'+variant+'-'+String(index).padStart(2,'0')+'.png';
}
function draw(){
 if(current.length!==2)return;
 const w=1536/zoom,h=864/zoom,x=Math.max(0,Math.min(1536-w,cx*1536-w/2)),y=Math.max(0,Math.min(864-h,cy*864-h/2));
 canvases.forEach((c,i)=>{const ctx=c.getContext('2d');ctx.clearRect(0,0,c.width,c.height);ctx.drawImage(current[i],x,y,w,h,0,0,c.width,c.height)});
 document.querySelectorAll('[data-zoom]').forEach(b=>b.setAttribute('aria-pressed',Number(b.dataset.zoom)===zoom));
}
function label(value,filter){return value==='aa'?'Точный эталон · 16 лучей на пиксель':'Полярная карта '+Number(value)+' см · '+(filter==='fixed'?'ширина края 5 см':'прежнее сглаживание')}
async function show(n,reset=false){
 const ticket=++serial,c=config(),urls=[file(c,'left',n),file(c,'right',n)];document.documentElement.dataset.ready='loading';
 if(reset){current=[];currentUrls=[];canvases.forEach(c=>c.getContext('2d').clearRect(0,0,c.width,c.height));status.textContent='Загрузка пары кадров…';}
 try{
  const pair=await Promise.all(urls.map(bitmap));if(ticket!==serial)return false;
  current=pair;currentUrls=urls;frame=n;draw();controls.time.value=n;
  document.querySelector('#leftLabel').textContent=label(c.left,c.filter);document.querySelector('#rightLabel').textContent=label(c.right,c.filter);
  document.querySelector('#description').textContent=c.track.description;
  const source=c.track.positions[n];document.querySelector('#frameLabel').textContent=(n+1)+' / '+report.frames+' · лампа ('+source.x.toFixed(3)+'; '+source.y.toFixed(3)+') м';
  shown.push(performance.now());if(shown.length>16)shown.shift();
  status.textContent='Оба кадра готовы.'+(playing&&shown.length>1?' Показ: '+((shown.length-1)*1000/(shown.at(-1)-shown[0])).toFixed(1)+' кадр/с.':'');
  document.documentElement.dataset.ready='true';trim();
  for(let ahead=1;ahead<=2;ahead++){const f=(n+ahead)%report.frames;bitmap(file(c,'left',f));bitmap(file(c,'right',f));}
  return true;
 }catch(error){if(ticket===serial){pause();current=[];currentUrls=[];canvases.forEach(c=>c.getContext('2d').clearRect(0,0,c.width,c.height));status.textContent=error.message;document.documentElement.dataset.ready='false';}return false;}
}
function pause(){playing=false;serial++;clearTimeout(timer);controls.play.textContent='Играть';shown=[];if(current.length===2)document.documentElement.dataset.ready='true';}
async function tick(){
 if(!playing)return;const start=performance.now(),ok=await show((frame+1)%report.frames);
 if(ok&&playing)timer=setTimeout(tick,Math.max(0,1000/(report.fps*Number(controls.speed.value))-(performance.now()-start)));
}
function reset(){pause();const flash=config().track.flash;if(flash){controls.left.value='aa';controls.right.value='05';controls.filter.value='fixed';}controls.filter.disabled=flash;controls.left.disabled=flash;controls.right.disabled=flash;show(frame,true);}
controls.play.onclick=()=>{if(playing){pause();return;}playing=true;controls.play.textContent='Пауза';shown=[];tick();};
controls.previous.onclick=()=>{pause();show((frame+report.frames-1)%report.frames)};
controls.next.onclick=()=>{pause();show((frame+1)%report.frames)};
controls.time.oninput=()=>{pause();show(Number(controls.time.value))};
for(const c of [controls.track,controls.filter,controls.left,controls.right])c.onchange=reset;
controls.speed.onchange=()=>{shown=[];};
document.querySelectorAll('[data-zoom]').forEach(b=>b.onclick=()=>{zoom=Number(b.dataset.zoom);draw()});
canvases.forEach(c=>c.onclick=e=>{const r=c.getBoundingClientRect(),w=1/zoom,h=1/zoom;cx=Math.max(0,Math.min(1-w,cx-w/2))+(e.clientX-r.left)/r.width*w;cy=Math.max(0,Math.min(1-h,cy-h/2))+(e.clientY-r.top)/r.height*h;draw();});
document.addEventListener('visibilitychange',()=>{if(document.hidden)pause()});
fetch('motion-report.json').then(r=>{if(!r.ok)throw Error('Нет отчёта движения');return r.json()}).then(async data=>{
 report=data;
 for(const track of report.tracks){const option=document.createElement('option');option.value=track.id;option.textContent=track.title;controls.track.append(option);}
 for(const c of [controls.track,controls.filter,controls.left,controls.right,controls.play,controls.previous,controls.next,controls.time])c.disabled=false;
 const timing=name=>report.timings.find(t=>t.operation===name),ms=t=>t.medianMs.toFixed(3)+' мс';
 for(const step of ['02','03','05']){const tr=document.createElement('tr'),moving=timing('moving-total-'+step);for(const value of [Number(step)+' см',ms(timing('build-'+step)),ms(timing('cached-frame-'+step)),ms(moving)+' / '+moving.p90Ms.toFixed(3)+' мс']){const td=document.createElement('td');td.textContent=value;tr.append(td);}document.querySelector('#timings').append(tr);}
 document.querySelector('#flashTiming').textContent='Вспышка: подготовка и показ '+ms(timing('flash-on-total-05'))+'; выключение '+ms(timing('flash-off'))+'.';
 document.querySelector('#method').textContent=report.method;document.querySelector('#validation').textContent=report.validation;
 document.querySelector('#device').textContent=report.device+' · '+report.api+' · GC '+report.gc0+'/'+report.gc1+'/'+report.gc2;
 await show(0,true);
}).catch(error=>{status.textContent=error.message;document.documentElement.dataset.ready='false'});
